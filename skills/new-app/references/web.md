# Web (React Router)

A single-page app: React Router v8 in framework mode with `ssr: false`, TypeScript, Mantine v9, and Vitest, using Yarn 4 through corepack.
Sign-in uses `react-oidc-context` against Cognito managed login (authorization code + PKCE). The API client is generated by Orval.
Config comes from `VITE_*` env vars that Vite writes into the bundle at build time. There is no runtime config file.

## Config and auth

```ts
// app/lib/auth/config.ts
export interface AppConfig {
  region: string;
  userPoolId: string;
  domain: string;
  clientId: string;
  redirectUri: string;
  logoutUri: string;
  apiUrl?: string;
}

type Env = Record<string, string | boolean | undefined>;

const REQUIRED_VARS = [
  "VITE_REGION",
  "VITE_USER_POOL_ID",
  "VITE_COGNITO_DOMAIN",
  "VITE_CLIENT_ID",
  "VITE_REDIRECT_URI",
  "VITE_LOGOUT_URI",
] as const;

function optional(value: string | boolean | undefined): string | undefined {
  return typeof value === "string" && value !== "" ? value : undefined;
}

// Reads config from VITE_* env vars, which Vite inlines at build time. Locally
// they come from .env.local; the deploy pipeline sets them for `yarn build`.
export function readAppConfig(env: Env = import.meta.env): AppConfig {
  const missing = REQUIRED_VARS.filter((name) => !optional(env[name]));
  if (missing.length > 0) {
    throw new Error(`Missing environment variables: ${missing.join(", ")}`);
  }

  return {
    region: env.VITE_REGION as string,
    userPoolId: env.VITE_USER_POOL_ID as string,
    domain: env.VITE_COGNITO_DOMAIN as string,
    clientId: env.VITE_CLIENT_ID as string,
    redirectUri: env.VITE_REDIRECT_URI as string,
    logoutUri: env.VITE_LOGOUT_URI as string,
    apiUrl: optional(env.VITE_API_URL),
  };
}
```

```tsx
// app/lib/config/app-config-context.tsx
import { createContext, useContext, useEffect, useState, type ReactNode } from "react";
import { readAppConfig, type AppConfig } from "~/lib/auth/config";

const AppConfigContext = createContext<AppConfig | undefined>(undefined);

// Reads config from build-time env vars and gates rendering of children until
// after mount. Because config is set in an effect, children never render during
// prerender or the initial (pre-hydration) client pass. Providers further down the tree (e.g. AppAuthProvider) rely on
// that ordering to avoid touching browser-only APIs too early.
export function AppConfigProvider({ children }: { children: ReactNode }) {
  const [config, setConfig] = useState<AppConfig | null>(null);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    try {
      setConfig(readAppConfig());
    } catch (err) {
      setError(err as Error);
    }
  }, []);

  if (error) {
    return <div style={{ padding: "2rem" }}>Failed to load application configuration: {error.message}</div>;
  }
  if (!config) {
    return null;
  }
  return <AppConfigContext.Provider value={config}>{children}</AppConfigContext.Provider>;
}

export function useAppConfig(): AppConfig {
  const config = useContext(AppConfigContext);
  if (config === undefined) {
    throw new Error("useAppConfig must be used within an AppConfigProvider");
  }
  return config;
}
```

```ts
// app/lib/auth/cognito.ts
import type { AuthProviderProps } from "react-oidc-context";
import type { User, UserManagerSettings } from "oidc-client-ts";
import type { AppConfig } from "./config";

function buildCognitoOidcSettings(config: AppConfig): UserManagerSettings {
  return {
    authority: `https://cognito-idp.${config.region}.amazonaws.com/${config.userPoolId}`,
    client_id: config.clientId,
    redirect_uri: config.redirectUri,
    post_logout_redirect_uri: config.logoutUri,
    response_type: "code",
    // aws.cognito.signin.user.admin lets the API read the email and delete the account with this token.
    scope: "aws.cognito.signin.user.admin email openid phone profile",
  };
}

export function buildCognitoOidcConfig(config: AppConfig): AuthProviderProps {
  return {
    ...buildCognitoOidcSettings(config),
    onSigninCallback: () => {
      window.history.replaceState({}, document.title, window.location.pathname);
    },
  };
}

// Same PKCE sign-in as the main AuthProvider, but opens managed login on the
// sign-up page. The request state lands in the shared sessionStorage store, so
// the main AuthProvider completes the callback on redirect_uri as usual.
export async function cognitoSignUpRedirect(config: AppConfig) {
  const { UserManager } = await import("oidc-client-ts");
  const userManager = new UserManager({
    ...buildCognitoOidcSettings(config),
    metadataSeed: { authorization_endpoint: `https://${config.domain}/signup` },
  });
  await userManager.signinRedirect();
}

// Cognito's /logout clears the managed-login session cookie; without it the next
// sign-in silently reuses the old session.
export function cognitoSignOutRedirect(config: AppConfig) {
  const params = new URLSearchParams({ client_id: config.clientId, logout_uri: config.logoutUri });
  window.location.href = `https://${config.domain}/logout?${params.toString()}`;
}

export function getUserDisplayName(user: User | null | undefined): string | undefined {
  return (user?.profile.email as string | undefined) ?? user?.profile.sub;
}
```

```tsx
// app/lib/auth/auth-provider.tsx
import { AuthProvider as OidcAuthProvider } from "react-oidc-context";
import { buildCognitoOidcConfig } from "./cognito";
import { useAppConfig } from "~/lib/config/app-config-context";

// react-oidc-context/oidc-client-ts touch browser-only APIs (window, sessionStorage)
// at construction time, so this must never be instantiated during prerender or on the
// initial (pre-hydration) client render. AppConfigProvider (an ancestor) only
// renders its children once config has loaded client-side, which guarantees that.
export function AppAuthProvider({ children }: { children: React.ReactNode }) {
  const config = useAppConfig();
  return <OidcAuthProvider {...buildCognitoOidcConfig(config)}>{children}</OidcAuthProvider>;
}
```

`root.tsx` uses the standard React Router `Layout` with Mantine (`ColorSchemeScript`, `mantineHtmlProps`, `MantineProvider`,
and an import of `@mantine/core/styles.css`) and the standard `ErrorBoundary`. The app component nests the providers in this order:

```tsx
export default function App() {
  return (
    <AppConfigProvider>
      <AppAuthProvider>
        <Outlet />
      </AppAuthProvider>
    </AppConfigProvider>
  );
}
```

`app/lib/auth/return-to.ts`: Cognito always returns to the one configured `redirect_uri`. So before redirecting to sign-in, the app
saves `pathname + search` in sessionStorage under `<app>.returnTo` (`rememberReturnTo`). After sign-in it reads and clears it
(`consumeReturnTo`). Wrap both in try/catch, because storage can be unavailable.

```ts

```

```ts
// app/lib/api/axios-instance.ts
import axios, { type AxiosRequestConfig } from "axios";

export interface IRequestOptions {
  accessToken?: string;
}

export const axiosInstance = axios.create({ baseURL: import.meta.env.VITE_API_URL });

// Every generated call takes { accessToken } as its last argument.
export const orvalMutator = <T>(config: AxiosRequestConfig, options?: IRequestOptions): Promise<T> => {
  const source = axios.CancelToken.source();
  const promise = axiosInstance({
    ...config,
    cancelToken: source.token,
    // Keep the generated headers (Content-Type for JSON and multipart bodies) and add the token.
    headers: { ...config.headers, Authorization: `Bearer ${options?.accessToken}` },
  }).then((response) => response.data);

  // @ts-ignore
  promise.cancel = () => source.cancel("Query was cancelled");
  return promise;
};
```

`app/lib/api/errors.ts` turns an axios error into a message a user can read. It uses the response body when the API sent a string
(the `BadRequest(message)` results) and a generic message otherwise.

Run `yarn orval` after every API change. Commit `generated/` and never edit it by hand. Orval reads the mutator file, so
**write `axios-instance.ts` before the first `yarn orval`**. Otherwise Orval fails with ENOENT.

Images are served by the API behind the bearer token, so an `<img src>` can't load them directly. `app/lib/api/images.ts` fetches
the image with `axiosInstance.get(path, { responseType: "blob", headers: { Authorization } })` and returns `URL.createObjectURL(blob)`.
Components revoke the object URL when they unmount or the path changes.

## Routes and the signed-in shell

```ts
// app/routes.ts
import { type RouteConfig, index, layout, prefix, route } from "@react-router/dev/routes";

export default [
  index("routes/home.tsx"),
  route("privacy", "routes/privacy.tsx"),
  route("terms", "routes/terms.tsx"),
  route("delete-account", "routes/delete-account.tsx"),
  route("signup", "routes/signup.tsx"),
  route("logout", "routes/logout.tsx"),
  layout("routes/authn-required/layout.tsx", [
    ...prefix("manage", [
      index("routes/authn-required/manage/dashboard.tsx"),
      route("profile", "routes/authn-required/manage/profile.tsx"),
    ]),
  ]),
] satisfies RouteConfig;
```

| Route | Content |
| --- | --- |
| `home` | Public landing page: the app name, a placeholder one-line pitch, and Sign in / Sign up buttons. Signed-in users go to `/manage` |
| `privacy`, `terms` | Public placeholder pages with a visible `TODO: legal review` banner. The app stores need them reachable without signing in |
| `delete-account` | Public page explaining how to delete the account (Google Play requires a public URL), with a link into Profile for signed-in users |
| `signup` | Calls `cognitoSignUpRedirect` |
| `logout` | `auth.removeUser()`, then `cognitoSignOutRedirect` |
| `manage/dashboard` | A placeholder welcome page for features to grow into |
| `manage/profile` | Shows the email, edits the name (`PUT /users/me`), and deletes the account (`DELETE /users/me` after a confirm modal, then logs out) |

`routes/authn-required/layout.tsx` is wrapped in `withAuthenticationRequired` from `react-oidc-context`, which redirects to sign-in and calls
`rememberReturnTo` first. It does this:
1. **Terms gate.** Users agree to the Terms of Use before anything else, including registering with the API. `lib/terms.ts` keeps
   acceptance in localStorage under a key that includes the user's `sub` and the terms version. React state covers browsers where
   localStorage is unavailable. Until the user agrees, render `<TermsAcceptance onAgree=... />` and nothing else.
2. **Register.** Once terms are accepted, call `postUsers({ accessToken })` and keep the `UserResponse` in state. Key the effect on
   `auth.user?.profile.sub`, not on the token, so silent token renewals don't register the user again.
3. **Return.** On mount, `consumeReturnTo()` and `navigate(returnTo, { replace: true })` if it differs from the current path.
4. **Shell.** A Mantine `AppShell` with the app name or logo linking to `/`, an account menu (Dashboard, Profile, Privacy Policy,
   Terms of Use, Sign out), icons only on mobile (`useMediaQuery("(max-width: 48em)")`), and an `Outlet` that gets
   `{ currentUser, setCurrentUser }` as `ManageOutletContext`.

To test `orvalMutator`, replace `axiosInstance.defaults.adapter` with a mock and restore it afterwards. Spying on `axiosInstance.request`
doesn't work, because calling `axiosInstance(config)` doesn't go through that property.

Write Vitest + Testing Library tests for `readAppConfig` (missing vars are listed, optional vars treat empty as undefined),
`AppConfigProvider` (an error message, children only after config loads), `orvalMutator` (sets the bearer header), `terms.ts`, `return-to.ts`,
and the profile page (saving the name, deleting after confirmation). Mock `react-oidc-context` and the generated API module.

**Done when:** `yarn install`, `yarn orval`, `yarn typecheck`, `yarn test` and `yarn build` all succeed.
